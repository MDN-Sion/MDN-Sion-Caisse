-- DÉFINITIF — à conserver. -- Ne touche ni aux éditions ni aux stocks opérationnels Réels. BEGIN; ALTER TABLE public.lignes_inventaires_editions_demo ADD COLUMN IF NOT EXISTS conserver_catalogue boolean NOT NULL DEFAULT true, ADD COLUMN IF NOT EXISTS actif_edition_suivante boolean NOT NULL DEFAULT true; CREATE OR REPLACE FUNCTION public.enregistrer_ligne_inventaire_options_demo( p_preparation_id uuid, p_categorie text, p_article text, p_quantite integer, p_conserver_catalogue boolean, p_actif_edition_suivante boolean ) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path TO '' AS $fonction$ DECLARE v_acces jsonb; v_article text; v_existe boolean; BEGIN v_acces := public.etat_acces_administration(); IF COALESCE((v_acces ->> 'acces_administration')::boolean, false) IS DISTINCT FROM true OR COALESCE((v_acces ->> 'mode_demo')::boolean, false) IS DISTINCT FROM true THEN RAISE EXCEPTION 'ACCES_DEMO_REQUIS'; END IF; IF p_preparation_id IS NULL OR NOT EXISTS ( SELECT 1 FROM public.preparations_editions_demo AS p WHERE p.id = p_preparation_id AND p.statut = 'brouillon' ) THEN RAISE EXCEPTION 'PREPARATION_DEMO_INTROUVABLE_OU_NON_MODIFIABLE'; END IF; v_article := btrim(COALESCE(p_article, '')); IF p_categorie IS NULL OR p_categorie NOT IN ('mdn', 'fondation', 'vaisselle', 'contenants') OR length(v_article) NOT BETWEEN 1 AND 150 OR p_quantite IS NULL OR p_quantite < 0 OR (p_categorie = 'fondation' AND p_quantite <> 0) OR p_conserver_catalogue IS NULL OR p_actif_edition_suivante IS NULL OR (NOT p_conserver_catalogue AND p_actif_edition_suivante) THEN RAISE EXCEPTION 'INVENTAIRE_DEMO_INVALIDE'; END IF; SELECT EXISTS ( SELECT 1 FROM ( SELECT CASE WHEN lower(btrim(a.beneficiaire)) = 'fondation' THEN 'fondation' ELSE 'mdn' END AS categorie, a.nom FROM public.referentiel_articles AS a WHERE a.actif IS TRUE AND lower(btrim(COALESCE(a.beneficiaire, ''))) IN ( 'fondation', 'mdn', 'marché', 'marche' ) UNION ALL SELECT 'vaisselle', v.nom FROM public.referentiel_vaisselle_mdn AS v WHERE v.actif IS TRUE UNION ALL SELECT 'contenants', c.nom FROM public.types_contenants AS c WHERE c.actif IS TRUE ) AS catalogue WHERE catalogue.categorie = p_categorie AND catalogue.nom = v_article ) INTO v_existe; IF NOT v_existe THEN RAISE EXCEPTION 'ARTICLE_ABSENT_DU_CATALOGUE'; END IF; INSERT INTO public.lignes_inventaires_editions_demo ( preparation_id, categorie, article, quantite, commentaire, conserver_catalogue, actif_edition_suivante ) VALUES ( p_preparation_id, p_categorie, v_article, p_quantite, '', p_conserver_catalogue, p_actif_edition_suivante ) ON CONFLICT (preparation_id, categorie, article) DO UPDATE SET quantite = EXCLUDED.quantite, conserver_catalogue = EXCLUDED.conserver_catalogue, actif_edition_suivante = EXCLUDED.actif_edition_suivante, updated_at = now(); DELETE FROM public.validations_inventaires_editions_demo AS v WHERE v.preparation_id = p_preparation_id AND v.categorie = p_categorie; RETURN jsonb_build_object( 'ok', true, 'mode_demo', true, 'categorie', p_categorie, 'article', v_article, 'validation_reinitialisee', true, 'stocks_reels_modifies', false ); END; $fonction$; CREATE OR REPLACE FUNCTION public.charger_inventaires_preparation_demo(p_preparation_id uuid) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path TO '' AS $fonction$ DECLARE v_acces jsonb; v_lignes jsonb; v_validations jsonb; BEGIN v_acces := public.etat_acces_administration(); IF COALESCE((v_acces ->> 'acces_administration')::boolean, false) IS DISTINCT FROM true OR COALESCE((v_acces ->> 'mode_demo')::boolean, false) IS DISTINCT FROM true THEN RAISE EXCEPTION 'ACCES_DEMO_REQUIS'; END IF; IF NOT EXISTS ( SELECT 1 FROM public.preparations_editions_demo WHERE id = p_preparation_id ) THEN RAISE EXCEPTION 'PREPARATION_DEMO_INTROUVABLE'; END IF; SELECT COALESCE( jsonb_agg( jsonb_build_object( 'id', l.id, 'categorie', l.categorie, 'article', l.article, 'quantite', l.quantite, 'commentaire', l.commentaire, 'conserver_catalogue', l.conserver_catalogue, 'actif_edition_suivante', l.actif_edition_suivante ) ORDER BY l.categorie, l.article ), '[]'::jsonb ) INTO v_lignes FROM public.lignes_inventaires_editions_demo AS l WHERE l.preparation_id = p_preparation_id; SELECT COALESCE( jsonb_object_agg(v.categorie, v.valide_at), '{}'::jsonb ) INTO v_validations FROM public.validations_inventaires_editions_demo AS v WHERE v.preparation_id = p_preparation_id; RETURN jsonb_build_object( 'ok', true, 'mode_demo', true, 'preparation_id', p_preparation_id, 'lignes', v_lignes, 'validations', v_validations ); END; $fonction$; CREATE OR REPLACE FUNCTION public.valider_inventaire_preparation_demo( p_preparation_id uuid, p_categorie text ) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path TO '' AS $fonction$ DECLARE v_acces jsonb; v_attendus integer; v_enregistres integer; BEGIN v_acces := public.etat_acces_administration(); IF COALESCE((v_acces ->> 'acces_administration')::boolean, false) IS DISTINCT FROM true OR COALESCE((v_acces ->> 'mode_demo')::boolean, false) IS DISTINCT FROM true THEN RAISE EXCEPTION 'ACCES_DEMO_REQUIS'; END IF; IF p_categorie IS NULL OR p_categorie NOT IN ('mdn', 'fondation', 'vaisselle', 'contenants') THEN RAISE EXCEPTION 'CATEGORIE_INVALIDE'; END IF; IF NOT EXISTS ( SELECT 1 FROM public.preparations_editions_demo AS p WHERE p.id = p_preparation_id AND p.statut = 'brouillon' ) THEN RAISE EXCEPTION 'PREPARATION_DEMO_INTROUVABLE_OU_NON_MODIFIABLE'; END IF; SELECT count(*) INTO v_attendus FROM ( SELECT CASE WHEN lower(btrim(a.beneficiaire)) = 'fondation' THEN 'fondation' ELSE 'mdn' END AS categorie, a.nom FROM public.referentiel_articles AS a WHERE a.actif IS TRUE AND lower(btrim(COALESCE(a.beneficiaire, ''))) IN ( 'fondation', 'mdn', 'marché', 'marche' ) UNION ALL SELECT 'vaisselle', v.nom FROM public.referentiel_vaisselle_mdn AS v WHERE v.actif IS TRUE UNION ALL SELECT 'contenants', c.nom FROM public.types_contenants AS c WHERE c.actif IS TRUE ) AS catalogue WHERE catalogue.categorie = p_categorie; SELECT count(*) INTO v_enregistres FROM public.lignes_inventaires_editions_demo AS l WHERE l.preparation_id = p_preparation_id AND l.categorie = p_categorie; IF v_attendus = 0 OR v_enregistres <> v_attendus THEN RAISE EXCEPTION 'INVENTAIRE_INCOMPLET_%_SUR_%', v_enregistres, v_attendus; END IF; IF EXISTS ( SELECT 1 FROM public.lignes_inventaires_editions_demo AS l WHERE l.preparation_id = p_preparation_id AND l.categorie = p_categorie AND ( (NOT l.conserver_catalogue AND l.actif_edition_suivante) OR (p_categorie = 'fondation' AND l.quantite <> 0) ) ) THEN RAISE EXCEPTION 'INVENTAIRE_OPTIONS_INCOHERENTES'; END IF; INSERT INTO public.validations_inventaires_editions_demo ( preparation_id, categorie, valide_par ) VALUES ( p_preparation_id, p_categorie, auth.uid() ) ON CONFLICT (preparation_id, categorie) DO UPDATE SET valide_at = now(), valide_par = EXCLUDED.valide_par; RETURN jsonb_build_object( 'ok', true, 'mode_demo', true, 'categorie', p_categorie, 'valide', true, 'articles_valides', v_enregistres, 'articles_attendus', v_attendus, 'stocks_reels_modifies', false ); END; $fonction$; COMMIT;
