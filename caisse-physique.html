-- SQL223 — DÉFINITIF — à conserver -- Consultation sécurisée du stock initial, des mouvements et du solde par édition Démo. -- Aucune écriture, aucune activation, aucun accès aux données Réelles. BEGIN; CREATE OR REPLACE FUNCTION public.consulter_stocks_operations_edition_demo( p_edition_demo_id uuid ) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $function$ DECLARE v_acces jsonb; v_edition public.editions_creees_demo%ROWTYPE; v_stocks jsonb; v_nombre_operations bigint; v_nombre_mouvements bigint; v_stocks_negatifs bigint; BEGIN v_acces := public.etat_acces_administration(); IF COALESCE((v_acces ->> 'acces_administration')::boolean, false) IS DISTINCT FROM true OR COALESCE((v_acces ->> 'mode_demo')::boolean, false) IS DISTINCT FROM true OR NOT EXISTS ( SELECT 1 FROM public.utilisateurs AS u WHERE u.auth_user_id = auth.uid() AND u.actif IS TRUE AND u.is_demo IS TRUE AND lower(btrim(u.role)) = 'administrateur' ) THEN RAISE EXCEPTION 'ADMINISTRATEUR_DEMO_REQUIS'; END IF; IF p_edition_demo_id IS NULL THEN RAISE EXCEPTION 'EDITION_DEMO_OBLIGATOIRE'; END IF; SELECT e.* INTO v_edition FROM public.editions_creees_demo AS e WHERE e.id = p_edition_demo_id; IF NOT FOUND THEN RAISE EXCEPTION 'EDITION_DEMO_INTROUVABLE'; END IF; -- Chaque solde est calculé exclusivement sur les mouvements de la même édition. WITH soldes AS ( SELECT s.id, s.categorie, s.article, s.quantite AS stock_initial, s.conserver_catalogue, s.actif_edition_suivante, COALESCE(SUM(m.quantite_signee), 0)::bigint AS mouvement_net, s.quantite::bigint + COALESCE(SUM(m.quantite_signee), 0)::bigint AS solde FROM public.stocks_editions_creees_demo AS s LEFT JOIN public.mouvements_stocks_editions_creees_demo AS m ON m.stock_edition_demo_id = s.id AND m.edition_demo_id = s.edition_demo_id WHERE s.edition_demo_id = p_edition_demo_id GROUP BY s.id, s.categorie, s.article, s.quantite, s.conserver_catalogue, s.actif_edition_suivante ) SELECT COALESCE( jsonb_agg( jsonb_build_object( 'stock_id', id, 'categorie', categorie, 'article', article, 'stock_initial', stock_initial, 'mouvement_net', mouvement_net, 'solde', solde, 'conserver_catalogue', conserver_catalogue, 'actif', actif_edition_suivante ) ORDER BY categorie, article ), '[]'::jsonb ), COUNT(*) FILTER (WHERE solde < 0) INTO v_stocks, v_stocks_negatifs FROM soldes; SELECT COUNT(*) INTO v_nombre_operations FROM public.operations_editions_creees_demo AS o WHERE o.edition_demo_id = p_edition_demo_id; SELECT COUNT(*) INTO v_nombre_mouvements FROM public.mouvements_stocks_editions_creees_demo AS m WHERE m.edition_demo_id = p_edition_demo_id; RETURN jsonb_build_object( 'ok', true, 'lecture_seule', true, 'mode_demo', true, 'edition_id', v_edition.id, 'annee', v_edition.annee, 'statut', v_edition.statut, 'est_active', v_edition.est_active, 'nombre_operations', v_nombre_operations, 'nombre_mouvements_stock', v_nombre_mouvements, 'nombre_stocks_negatifs', v_stocks_negatifs, 'stocks', v_stocks, 'moteur_caisse_raccorde', false, 'activation_autorisee_par_cette_fonction', false ); END; $function$; REVOKE ALL ON FUNCTION public.consulter_stocks_operations_edition_demo(uuid) FROM PUBLIC, anon; GRANT EXECUTE ON FUNCTION public.consulter_stocks_operations_edition_demo(uuid) TO authenticated; COMMIT;
